name: Refresh Research Terminal

on:
  schedule:
    - cron: '7 * * * *'
  workflow_dispatch:

concurrency:
  group: refresh-terminal
  cancel-in-progress: false

jobs:
  build:
    runs-on: ubuntu-latest
    permissions:
      contents: write
    steps:
      - uses: actions/checkout@v4.2.2
        with:
          fetch-depth: 0
      - uses: actions/setup-python@v5.6.0
        with:
          python-version: '3.11'
      - run: pip install yfinance pandas numpy --quiet
      - name: Rebuild terminal
        run: python scripts/terminal_test.py
      - name: Commit and push terminal
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "github-actions[bot]@users.noreply.github.com"
          git add terminal
          if git diff --staged --quiet; then echo "terminal unchanged"; exit 0; fi
          git commit -m "terminal: hourly rebuild [skip ci]"
          git pull --rebase origin main
          git push
